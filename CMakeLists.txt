cmake_minimum_required(VERSION 3.20)
project(Editor3D LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
 set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(vendor/glfw)
find_package(OpenGL REQUIRED)
add_library(imgui vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/backends/imgui_impl_glfw.cpp vendor/imgui/backends/imgui_impl_opengl2.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)
add_executable(Editor3D src/main.cpp src/Scene.cpp src/Mesh.cpp src/Renderer.cpp src/Editor.cpp)
target_include_directories(Editor3D PRIVATE src vendor/glm vendor/json/single_include)
target_link_libraries(Editor3D PRIVATE imgui)
target_compile_definitions(Editor3D PRIVATE ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(MSVC)
 target_compile_options(Editor3D PRIVATE /W4 /utf-8)
else()
 target_compile_options(Editor3D PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(CoreTests tests/core_tests.cpp src/Scene.cpp src/Mesh.cpp)
target_include_directories(CoreTests PRIVATE src vendor/glm vendor/json/single_include)
enable_testing()
add_test(NAME scene_math_persistence COMMAND CoreTests)
